```sql
-- === Script 1: Record Count Parity Check ===

WITH source_counts AS (
  SELECT 
    'customers' AS table_name, COUNT(*) AS rec_count 
    FROM <source_schema>.customers
  UNION ALL
  SELECT 'orders', COUNT(*) 
    FROM <source_schema>.orders
  UNION ALL
  SELECT 'transactions', COUNT(*) 
    FROM <source_schema>.transactions
  UNION ALL
  SELECT 'accounts', COUNT(*) 
    FROM <source_schema>.accounts
),
target_counts AS (
  SELECT 
    'customers' AS table_name, COUNT(*) AS rec_count 
    FROM <target_schema>.customers
  UNION ALL
  SELECT 'orders', COUNT(*) 
    FROM <target_schema>.orders
  UNION ALL
  SELECT 'transactions', COUNT(*) 
    FROM <target_schema>.transactions
  UNION ALL
  SELECT 'accounts', COUNT(*) 
    FROM <target_schema>.accounts
)
SELECT 
  sc.table_name,
  sc.rec_count AS source_count,
  tc.rec_count AS target_count,
  CASE 
    WHEN sc.rec_count = tc.rec_count THEN 'PASS' 
    ELSE 'FAIL: MISMATCH' 
  END AS verification_status
FROM source_counts sc
JOIN target_counts tc ON sc.table_name = tc.table_name
ORDER BY sc.table_name;
```
